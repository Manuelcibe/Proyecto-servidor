#!/bin/bash
# ═══════════════════════════════════════════════════════════════
# Hostify - Script de preparación de la estación de control
# Ejecuta esto antes de ansible-playbook
# ═══════════════════════════════════════════════════════════════

set -e

echo "═══════════════════════════════════════════════════════════════"
echo "  Hostify - Preparando estación de control para Ansible"
echo "═══════════════════════════════════════════════════════════════"

# Verificar que Ansible está instalado
if ! command -v ansible &> /dev/null; then
    echo "❌ Ansible no está instalado."
    echo "   Instálalo con: sudo apt install ansible"
    echo "   O: pip install ansible"
    exit 1
fi

echo "✅ Ansible detectado: $(ansible --version | head -1)"

# Instalar colecciones requeridas
echo ""
echo "📦 Instalando colecciones de Ansible Galaxy..."
ansible-galaxy collection install -r requirements.yml

# Verificar conectividad con el host
echo ""
echo "🔌 Probando conectividad SSH con el host..."
if ansible -i inventory.yml hostify -m ping; then
    echo "✅ Conectividad OK"
else
    echo "❌ No se puede conectar al host"
    echo "   Verifica:"
    echo "   - El contenedor está encendido"
    echo "   - La IP en inventory.yml es correcta"
    echo "   - SSH está habilitado para root"
    echo "   - Puedes hacer: ssh root@<IP>"
    exit 1
fi

echo ""
echo "═══════════════════════════════════════════════════════════════"
echo "  ✅ Todo listo. Ejecuta el playbook con:"
echo ""
echo "     ansible-playbook -i inventory.yml playbook.yml"
echo ""
echo "═══════════════════════════════════════════════════════════════"

---
# ═══════════════════════════════════════════════════════════════
# ROL 01: Sistema base
# Configura hostname, timezone, paquetes esenciales y Java 21
# ═══════════════════════════════════════════════════════════════

- name: Configurar hostname
  ansible.builtin.hostname:
    name: "{{ hostify_hostname }}"

- name: Añadir hostname a /etc/hosts
  ansible.builtin.lineinfile:
    path: /etc/hosts
    regexp: '^127\.0\.1\.1'
    line: "127.0.1.1 {{ hostify_hostname }}"
    state: present

- name: Configurar timezone
  community.general.timezone:
    name: "{{ timezone }}"

- name: Actualizar caché de paquetes apt
  ansible.builtin.apt:
    update_cache: yes
    cache_valid_time: 3600

- name: Instalar paquetes base (sin Java de momento)
  ansible.builtin.apt:
    name:
      - curl
      - wget
      - unzip
      - git
      - nano
      - sudo
      - software-properties-common
      - apt-transport-https
      - ca-certificates
      - gnupg
      - lsb-release
      - python3-pip
      - python3-pymysql
      - net-tools
      - dnsutils
      - htop
    state: present

# ──────────────────────────────────────────
# Java 21 desde Adoptium (Eclipse Temurin)
# Debian 12 no tiene Java 21 en repos oficiales
# ──────────────────────────────────────────
- name: Crear directorio /etc/apt/keyrings
  ansible.builtin.file:
    path: /etc/apt/keyrings
    state: directory
    mode: '0755'

- name: Añadir clave GPG de Adoptium
  ansible.builtin.get_url:
    url: https://packages.adoptium.net/artifactory/api/gpg/key/public
    dest: /etc/apt/keyrings/adoptium.asc
    mode: '0644'

- name: Añadir repositorio de Adoptium
  ansible.builtin.copy:
    dest: /etc/apt/sources.list.d/adoptium.list
    content: "deb [signed-by=/etc/apt/keyrings/adoptium.asc] https://packages.adoptium.net/artifactory/deb {{ ansible_distribution_release }} main\n"
    mode: '0644'

- name: Actualizar caché tras añadir Adoptium
  ansible.builtin.apt:
    update_cache: yes

- name: Instalar Java 21 (Temurin)
  ansible.builtin.apt:
    name: temurin-21-jre
    state: present

- name: Verificar instalación de Java
  ansible.builtin.command: java -version
  register: java_version
  changed_when: false

- name: Mostrar versión de Java instalada
  ansible.builtin.debug:
    msg: "{{ java_version.stderr_lines }}"

# ──────────────────────────────────────────
# Directorios y locale
# ──────────────────────────────────────────
- name: Crear directorio de descargas temporales
  ansible.builtin.file:
    path: /tmp/hostify-downloads
    state: directory
    mode: '0755'

- name: Configurar locale en español
  ansible.builtin.shell: |
    if ! locale -a | grep -q "es_ES.utf8"; then
      sed -i 's/# es_ES.UTF-8/es_ES.UTF-8/' /etc/locale.gen
      locale-gen
      update-locale LANG=es_ES.UTF-8
    fi
  args:
    executable: /bin/bash
  register: locale_result
  changed_when: "'es_ES.utf8' not in locale_result.stdout"
  ignore_errors: yes

# ═══════════════════════════════════════════════════════════════
# ROL 02: Base de datos MariaDB
# ═══════════════════════════════════════════════════════════════

- name: Instalar MariaDB
  ansible.builtin.apt:
    name:
      - mariadb-server
      - mariadb-client
    state: present

- name: Iniciar y habilitar MariaDB
  ansible.builtin.systemd:
    name: mariadb
    enabled: yes
    state: started

- name: Esperar a que MariaDB esté disponible
  ansible.builtin.wait_for:
    port: 3306
    host: 127.0.0.1
    timeout: 30

- name: Configurar contraseña root MariaDB
  community.mysql.mysql_user:
    login_unix_socket: /var/run/mysqld/mysqld.sock
    name: root
    password: "{{ mariadb_root_password }}"
    host_all: yes
    state: present
  no_log: true

- name: Crear archivo .my.cnf para root (acceso sin password)
  ansible.builtin.copy:
    dest: /root/.my.cnf
    mode: '0600'
    content: |
      [client]
      user=root
      password={{ mariadb_root_password }}
  no_log: true

- name: Eliminar usuarios anónimos
  community.mysql.mysql_user:
    login_user: root
    login_password: "{{ mariadb_root_password }}"
    name: ''
    host_all: yes
    state: absent

- name: Eliminar base de datos test
  community.mysql.mysql_db:
    login_user: root
    login_password: "{{ mariadb_root_password }}"
    name: test
    state: absent

- name: Crear base de datos hostify
  community.mysql.mysql_db:
    login_user: root
    login_password: "{{ mariadb_root_password }}"
    name: "{{ nameless_db_name }}"
    encoding: utf8mb4
    collation: utf8mb4_unicode_ci
    state: present

- name: Crear usuario nameless con mysql_native_password
  ansible.builtin.shell: |
    mysql -u root -p'{{ mariadb_root_password }}' -e "
    CREATE USER IF NOT EXISTS '{{ nameless_db_user }}'@'localhost'
    IDENTIFIED VIA mysql_native_password USING PASSWORD('{{ nameless_db_password }}');
    GRANT ALL PRIVILEGES ON {{ nameless_db_name }}.* TO '{{ nameless_db_user }}'@'localhost';
    FLUSH PRIVILEGES;"
  no_log: true
  register: db_user_result
  changed_when: db_user_result.rc == 0

# ═══════════════════════════════════════════════════════════════
# ROL 03: Servidor web (Apache + PHP)
# ═══════════════════════════════════════════════════════════════

- name: Añadir repositorio de PHP (Sury)
  ansible.builtin.shell: |
    curl -sSLo /usr/share/keyrings/deb.sury.org-php.gpg https://packages.sury.org/php/apt.gpg
    echo "deb [signed-by=/usr/share/keyrings/deb.sury.org-php.gpg] https://packages.sury.org/php/ {{ ansible_distribution_release }} main" > /etc/apt/sources.list.d/php.list
  args:
    creates: /etc/apt/sources.list.d/php.list

- name: Actualizar caché tras añadir repo PHP
  ansible.builtin.apt:
    update_cache: yes

- name: Instalar Apache y PHP 8.3 con todas las extensiones
  ansible.builtin.apt:
    name:
      - apache2
      - libapache2-mod-php8.3
      - php8.3
      - php8.3-cli
      - php8.3-common
      - php8.3-mysql
      - php8.3-curl
      - php8.3-gd
      - php8.3-mbstring
      - php8.3-xml
      - php8.3-zip
      - php8.3-bcmath
      - php8.3-intl
      - php8.3-imagick
      - php8.3-soap
      - php8.3-readline
    state: present

- name: Habilitar módulos de Apache
  community.general.apache2_module:
    name: "{{ item }}"
    state: present
  loop:
    - rewrite
    - headers
    - ssl
  notify: restart apache

- name: Configurar Apache para NamelessMC
  ansible.builtin.copy:
    dest: /etc/apache2/sites-available/000-default.conf
    content: |
      <VirtualHost *:80>
          ServerName {{ hostify_hostname }}
          ServerAdmin {{ admin_email }}
          DocumentRoot /var/www/html

          <Directory /var/www/html>
              Options Indexes FollowSymLinks
              AllowOverride All
              Require all granted
          </Directory>

          ErrorLog ${APACHE_LOG_DIR}/error.log
          CustomLog ${APACHE_LOG_DIR}/access.log combined
      </VirtualHost>
    mode: '0644'
  notify: restart apache

- name: Aumentar límites de PHP
  ansible.builtin.lineinfile:
    path: /etc/php/8.3/apache2/php.ini
    regexp: "{{ item.regexp }}"
    line: "{{ item.line }}"
  loop:
    - { regexp: '^upload_max_filesize', line: 'upload_max_filesize = 100M' }
    - { regexp: '^post_max_size', line: 'post_max_size = 100M' }
    - { regexp: '^memory_limit', line: 'memory_limit = 256M' }
    - { regexp: '^max_execution_time', line: 'max_execution_time = 300' }
    - { regexp: '^date.timezone', line: 'date.timezone = {{ timezone }}' }
  notify: restart apache

- name: Iniciar y habilitar Apache
  ansible.builtin.systemd:
    name: apache2
    enabled: yes
    state: started

# ═══════════════════════════════════════════════════════════════
# ROL 04: PufferPanel (vía packagecloud, lo que funciona)
# ═══════════════════════════════════════════════════════════════

- name: Comprobar si PufferPanel ya está instalado
  ansible.builtin.stat:
    path: /usr/sbin/pufferpanel
  register: pufferpanel_installed

- name: Instalar repositorio de PufferPanel desde packagecloud
  ansible.builtin.shell: |
    curl -s https://packagecloud.io/install/repositories/pufferpanel/pufferpanel/script.deb.sh | bash
  args:
    creates: /etc/apt/sources.list.d/pufferpanel_pufferpanel.list
  when: not pufferpanel_installed.stat.exists

- name: Actualizar caché tras añadir repo PufferPanel
  ansible.builtin.apt:
    update_cache: yes
  when: not pufferpanel_installed.stat.exists

- name: Instalar PufferPanel
  ansible.builtin.apt:
    name: pufferpanel
    state: present

- name: Habilitar y arrancar PufferPanel
  ansible.builtin.systemd:
    name: pufferpanel
    enabled: yes
    state: started

- name: Esperar a que PufferPanel esté disponible
  ansible.builtin.wait_for:
    port: "{{ pufferpanel_port }}"
    host: 127.0.0.1
    timeout: 60

# ──────────────────────────────────────────
# Crear usuario admin de PufferPanel
# ──────────────────────────────────────────
- name: Comprobar si el admin de PufferPanel ya existe
  ansible.builtin.shell: |
    pufferpanel user list 2>/dev/null | grep -q "{{ pufferpanel_admin_username }}"
  register: pp_user_exists
  failed_when: false
  changed_when: false

- name: Crear usuario admin de PufferPanel
  ansible.builtin.shell: |
    pufferpanel user add \
      --admin \
      --email "{{ pufferpanel_admin_email }}" \
      --name "{{ pufferpanel_admin_username }}" \
      --password "{{ pufferpanel_admin_password }}"
  when: pp_user_exists.rc != 0
  no_log: true

- name: Mostrar info de PufferPanel
  ansible.builtin.debug:
    msg:
      - "✅ PufferPanel instalado correctamente"
      - "   URL: http://{{ hostify_ip }}:{{ pufferpanel_port }}"
      - "   Usuario: {{ pufferpanel_admin_username }}"
