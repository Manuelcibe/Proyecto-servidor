#!/bin/bash
# =============================================================================
# Proxmox vzdump Hook Script — Subida automática a Google Drive con rclone
# Ruta: /usr/local/bin/rclone-gdrive-hook.sh
# =============================================================================

# ──────────────────────────────────────────────────────────────────────────────
# CONFIGURACIÓN — Ajusta estas variables a tu entorno
# ──────────────────────────────────────────────────────────────────────────────

RCLONE_REMOTE="gdrive:Proxmox-Backups"   # Remote de rclone + carpeta destino
RCLONE_CONFIG="/root/.config/rclone/rclone.conf"  # Ruta al config de rclone
LOG_FILE="/var/log/rclone-gdrive-upload.log"       # Log de subidas
# Nº de copias a mantener en GDrive por VMID (0 = sin límite, cuidado con costes)
MAX_BACKUPS_PER_VM=3
# Usar 'copy' (mantiene local) o 'move' (elimina local tras subir)
RCLONE_ACTION="copy"

# ──────────────────────────────────────────────────────────────────────────────
# LÓGICA DEL HOOK — No modificar salvo que sepas lo que haces
# ──────────────────────────────────────────────────────────────────────────────

PHASE="$1"    # Proxmox pasa la fase como primer argumento
VMID="$3"     # ID de la VM/CT como segundo argumento

# Función de logging con timestamp
log() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] [VMID:${VMID}] $1" | tee -a "$LOG_FILE"
}

# Solo actuamos en la fase de finalización exitosa del backup
if [ "$PHASE" != "backup-end" ]; then
    exit 0
fi

log "======================================================"
log "Fase 'backup-end' detectada. Iniciando subida a GDrive."

# Proxmox expone la ruta del backup en la variable de entorno TARFILE
# Si por algún motivo está vacía, la buscamos manualmente
if [ -z "$TARFILE" ]; then
    log "ADVERTENCIA: Variable TARFILE no definida. Buscando en DUMPDIR..."
    # DUMPDIR también es proporcionada por Proxmox
    if [ -z "$DUMPDIR" ]; then
        log "ERROR CRÍTICO: DUMPDIR tampoco está definida. Abortando."
        exit 1
    fi
    # Buscar el archivo más reciente del VMID en el directorio de dumps
    TARFILE=$(find "$DUMPDIR" -maxdepth 1 \
        -name "vzdump-*-${VMID}-*.vma.zst" \
        -o -name "vzdump-*-${VMID}-*.tar.zst" \
        -o -name "vzdump-*-${VMID}-*.vma.gz" \
        -o -name "vzdump-*-${VMID}-*.tar.gz" \
        2>/dev/null | sort -t '-' -k5,6 | tail -n 1)
fi

if [ -z "$TARFILE" ] || [ ! -f "$TARFILE" ]; then
    log "ERROR: No se encontró el archivo de backup para VMID ${VMID}. Abortando."
    exit 1
fi

FILENAME=$(basename "$TARFILE")
log "Archivo de backup localizado: $FILENAME"
log "Iniciando $RCLONE_ACTION → ${RCLONE_REMOTE}/"

# Ejecutar rclone con parámetros robustos para entorno de producción
rclone "$RCLONE_ACTION" "$TARFILE" "${RCLONE_REMOTE}/" \
    --config "$RCLONE_CONFIG" \
    --transfers 1 \
    --checkers 4 \
    --retries 3 \
    --retries-sleep 30s \
    --stats 60s \
    --log-file "$LOG_FILE" \
    --log-level INFO \
    2>&1

RCLONE_EXIT=$?

if [ $RCLONE_EXIT -eq 0 ]; then
    log "✓ Subida completada correctamente: $FILENAME"

    # ── Rotación de backups antiguos en Google Drive ──────────────────────────
    if [ "$MAX_BACKUPS_PER_VM" -gt 0 ]; then
        log "Aplicando retención: máximo ${MAX_BACKUPS_PER_VM} backups por VM en GDrive."

        # Listar archivos del VMID en GDrive ordenados por fecha (más antiguo primero)
        REMOTE_FILES=$(rclone lsf "${RCLONE_REMOTE}/" \
            --config "$RCLONE_CONFIG" \
            --include "vzdump-*-${VMID}-*" \
            --format "tp" \
            2>>"$LOG_FILE" | sort | awk '{print $2}')

        TOTAL=$(echo "$REMOTE_FILES" | grep -c '.' || true)
        DELETE_COUNT=$(( TOTAL - MAX_BACKUPS_PER_VM ))

        if [ "$DELETE_COUNT" -gt 0 ]; then
            log "Hay ${TOTAL} backups remotos. Eliminando ${DELETE_COUNT} más antiguo(s)..."
            echo "$REMOTE_FILES" | head -n "$DELETE_COUNT" | while read -r OLD_FILE; do
                log "  Eliminando de GDrive: $OLD_FILE"
                rclone deletefile "${RCLONE_REMOTE}/${OLD_FILE}" \
                    --config "$RCLONE_CONFIG" \
                    2>>"$LOG_FILE"
            done
        else
            log "Retención OK. No se eliminan archivos remotos (${TOTAL}/${MAX_BACKUPS_PER_VM})."
        fi
    fi
else
    log "✗ ERROR en la subida (código rclone: ${RCLONE_EXIT}). Revisa $LOG_FILE"
    # Salimos con código de error para que Proxmox lo registre como warning
    exit $RCLONE_EXIT
fi

log "Hook finalizado."
exit 0
