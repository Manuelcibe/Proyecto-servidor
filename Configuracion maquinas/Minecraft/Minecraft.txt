async function hostifySend() {
  const input = document.getElementById('hostify-chat-input');
  const messages = document.getElementById('hostify-chat-messages');
  const text = input.value.trim();
  if (!text) return;
  
  const userMsg = document.createElement('div');
  userMsg.className = 'hostify-msg user';
  userMsg.textContent = text;
  messages.appendChild(userMsg);
  input.value = '';
  
  const typing = document.createElement('div');
  typing.className = 'hostify-typing';
  typing.textContent = 'Escribiendo...';
  typing.id = 'hostify-typing';
  messages.appendChild(typing);
  messages.scrollTop = messages.scrollHeight;
  
  try {
    const res = await fetch('/gemini-proxy.php', {
      method: 'POST',
      headers: {'Content-Type': 'application/json'},
      body: JSON.stringify({message: text})
    });
    const data = await res.json();
    document.getElementById('hostify-typing').remove();
    
    const botMsg = document.createElement('div');
    botMsg.className = 'hostify-msg bot';
    botMsg.textContent = data.reply || 'Error al responder';
    messages.appendChild(botMsg);
    messages.scrollTop = messages.scrollHeight;
  } catch (e) {
    document.getElementById('hostify-typing').remove();
    const errMsg = document.createElement('div');
    errMsg.className = 'hostify-msg bot';
    errMsg.textContent = 'Error de conexión';
    messages.appendChild(errMsg);
  }
}
